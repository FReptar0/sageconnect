```javascript
/**
 * Formats an ISO timestamp as a Spanish relative time string.
 * Examples: 'hace 45s', 'hace 4m 12s', 'hace 2h 15m', 'hace 3d 4h'.
 * Returns 'No disponible' on invalid input.
 *
 * Used by Phase 17 'Operación en curso' card and heartbeat refresh ticker.
 * @param {string} isoString
 * @returns {string}
 */
function formatRelative(isoString) {
    if (!isoString) return 'No disponible';
    try {
        const past = new Date(isoString).getTime();
        if (Number.isNaN(past)) return 'No disponible';
        const diffMs = Date.now() - past;
        if (diffMs < 0) return 'en el futuro';

        const totalSeconds = Math.floor(diffMs / 1000);
        if (totalSeconds < 60) {
            return 'hace ' + totalSeconds + 's';
        }

        const totalMinutes = Math.floor(totalSeconds / 60);
        const seconds = totalSeconds % 60;
        if (totalMinutes < 60) {
            return 'hace ' + totalMinutes + 'm ' + seconds + 's';
        }

        const totalHours = Math.floor(totalMinutes / 60);
        const minutes = totalMinutes % 60;
        if (totalHours < 24) {
            return 'hace ' + totalHours + 'h ' + minutes + 'm';
        }

        const totalDays = Math.floor(totalHours / 24);
        const hours = totalHours % 24;
        return 'hace ' + totalDays + 'd ' + hours + 'h';
    } catch {
        return 'No disponible';
    }
}
```

         * Polls /api/operations/status, updates the "Operación en curso" card,
         * and (if an operationId is detected and no SSE is open yet) connects SSE
         * to feed the existing "Progreso en Tiempo Real" timeline. Hides the card
         * when no operations are active and closes SSE.
         *
         * Called every 5s by activeOpPollHandle (D-06). Also called once on page load.
         * Replaces the old detectRunningOperation() removed in Task 2a (had a bug:
         * operations.find() on a map always returned undefined — see Phase 17 D-04).
         */
        async function pollActiveOperation() {
            try {
                const res = await apiCall('GET', '/api/operations/status');
                if (!res || !res.success || !res.data || !res.data.operations) {
                    hideActiveOperationCard();
                    return;
                }

                // D-04 bug fix: operations is a MAP keyed by operationType, not an array.
                // Use direct map access (or Object.values when iterating) — never .find().
                const bgOp = res.data.operations['background-cycle'] || null;

                if (!bgOp || !bgOp.operationId) {
                    hideActiveOperationCard();
                    return;
                }

                activeOpSnapshot = bgOp;
                renderActiveOperationCard(bgOp);

                // D-05 hybrid: when polling detects an op AND no SSE is open, open one
                // so the existing timeline (Phase 9) gets fed. SSE never updates THIS card.
                if (!evtSource) {
                    connectSSE(bgOp.operationId);
                }
            } catch (err) {
                console.warn('Error polling /api/operations/status:', err);
                // Do NOT hide the card on transient network errors — leave the last snapshot.
            }
        }

        /**
         * Renders the active operation card from a /status response entry.
         * Per D-08, "step actual" = LAST stepProgress entry without finishedAt.
         * Per D-07, when stepProgress is empty, show "Inicializando ciclo…".
         */
        function renderActiveOperationCard(op) {
            const card = document.getElementById('active-operation-card');
            card.style.display = '';

            // D-10: truncate operationId to first 8 chars; full UUID in tooltip
            const idEl = document.getElementById('active-op-id');
            const shortId = (op.operationId || '').substring(0, 8) || '—';
            idEl.textContent = shortId;
            idEl.title = op.operationId || '';

            // D-09: relative elapsed since startedAt; full ISO in tooltip
            const elapsedEl = document.getElementById('active-op-elapsed');
            elapsedEl.textContent = formatRelative(op.startedAt);
            elapsedEl.title = op.startedAt || '';

            // D-07 / D-08: "step actual" line
            const stepEl = document.getElementById('active-op-step');
            const sp = Array.isArray(op.stepProgress) ? op.stepProgress : [];

            if (sp.length === 0) {
                stepEl.textContent = 'Inicializando ciclo…';
                stepEl.title = '';
                return;
            }

            // Last entry without finishedAt = "currently running"
            // If all are finished (e.g. inter-tenant pause), show last entry as "completado"
            let active = null;
            for (let i = sp.length - 1; i >= 0; i--) {
                if (!sp[i].finishedAt) { active = sp[i]; break; }
            }
            const last = sp[sp.length - 1];
            const display = active || last;
            const label = STEP_LABELS[display.step] || display.step;
            const tenantPart = display.tenant ? ' · ' + display.tenant : '';
            const heartbeatPart = ' · heartbeat ' + formatRelative(display.startedAt);
            const status = active ? '' : ' (completado)';
            stepEl.textContent = label + tenantPart + heartbeatPart + status;
            stepEl.title = 'startedAt: ' + (display.startedAt || '') + (display.finishedAt ? '  finishedAt: ' + display.finishedAt : '');
        }

        /**
         * Hides the active operation card and closes SSE if open. Called when polling
         * sees no active operations (cycle just finished or never started).
         */
        function hideActiveOperationCard() {
            const card = document.getElementById('active-operation-card');
            if (card) card.style.display = 'none';
            activeOpSnapshot = null;
            if (evtSource) {
                evtSource.close();
                evtSource = null;
            }
        }

        /**
         * Refreshes only the relative-time strings on the active operation card
         * WITHOUT calling the API. Driven by setInterval(1000) per D-08.
         * Reads activeOpSnapshot which was set by the last successful poll.
         */
        function refreshActiveOperationHeartbeat() {
            if (!activeOpSnapshot) return;
            const elapsedEl = document.getElementById('active-op-elapsed');
            if (elapsedEl) elapsedEl.textContent = formatRelative(activeOpSnapshot.startedAt);

            // Update step heartbeat without re-deriving "active" — same logic as renderActiveOperationCard
            const sp = Array.isArray(activeOpSnapshot.stepProgress) ? activeOpSnapshot.stepProgress : [];
            if (sp.length === 0) return;
            let active = null;
            for (let i = sp.length - 1; i >= 0; i--) {
                if (!sp[i].finishedAt) { active = sp[i]; break; }
            }
            const display = active || sp[sp.length - 1];
            const label = STEP_LABELS[display.step] || display.step;
            const tenantPart = display.tenant ? ' · ' + display.tenant : '';
            const heartbeatPart = ' · heartbeat ' + formatRelative(display.startedAt);
            const status = active ? '' : ' (completado)';
            const stepEl = document.getElementById('active-op-step');
            if (stepEl) stepEl.textContent = label + tenantPart + heartbeatPart + status;
        }
