  <name>Task 1: Create unified .env and documented .env.example</name>
  <files>.env, .env.example</files>
  <action>
    1. Read the ACTUAL values from the existing env files on disk:
       - `.env.credentials.focaltec` (this one exists with real values)
       - `.env.path` (this one exists with real values: PATH=./downloads, PROVIDERS_PATH=./downloads/providers, LOG_PATH=./logs)
       - Check for `.env.credentials.database` and `.env.credentials.mailing` (may or may not exist on disk)
       - Check for `.env` (the main one, may have real values)

    2. Create `.env` file in project root with ALL variables organized by sections:
       ```
       # ====== DATABASE ======
       USER=
       PASSWORD=
       SERVER=
       DATABASE=

       # ====== PORTAL (Focaltec) ======
       # For multiple tenants, separate values with commas
       URL=
       TENANT_ID=
       API_KEY=
       API_SECRET=
       DATABASES=
       EXTERNAL_IDS=

       # ====== MAILING ======
       # Transport type: smtp or gmail
       MAIL_TRANSPORT=smtp
       eFrom=
       ePass=
       eServer=
       ePuerto=587
       eSSL=TRUE
       MAILING_NOTICES=
       MAILING_CC=
       # Gmail OAuth (only if MAIL_TRANSPORT=gmail)
       CLIENT_ID=
       SECRET_CLIENT=
       REFRESH_TOKEN=
       REDIRECT_URI=

       # ====== PATHS ======
       PATH=./downloads
       PROVIDERS_PATH=./downloads/providers
       LOG_PATH=./logs

       # ====== APP ======
       IMPORT_CFDIS_ROUTE=
       ARG=
       NOMBRE=
       RFC=
       REGIMEN=
       TIMEZONE=America/Mexico_City
       # Default address for purchase orders
       DEFAULT_ADDRESS_CITY=
       DEFAULT_ADDRESS_COUNTRY=
       DEFAULT_ADDRESS_IDENTIFIER=
       DEFAULT_ADDRESS_MUNICIPALITY=
       DEFAULT_ADDRESS_STATE=
       DEFAULT_ADDRESS_STREET=
       DEFAULT_ADDRESS_ZIP=
       # Location identifiers to skip (comma-separated)
       ADDRESS_IDENTIFIERS_SKIP=
       # Auto-terminate after background processes complete (for scheduled tasks)
       AUTO_TERMINATE=false
       ```

       IMPORTANT: For variables that have real values in existing files, COPY those real values into the unified .env. For variables that don't have existing values, leave them empty or with sensible defaults as shown above.

    3. Create `.env.example` with the SAME structure but with example/placeholder values and more detailed inline comments:
       - Use section headers: `# ====== SECTION_NAME ======`
       - Add inline comments explaining each variable's purpose and expected format
       - For comma-separated vars, show multi-value example: `TENANT_ID=tenant1,tenant2`
       - For MAIL_TRANSPORT, document both options
       - For boolean vars, document accepted values
       - This file must be self-sufficient documentation -- an operator should understand every variable without reading other docs

    4. The .env.example should NOT contain real credentials -- only descriptive placeholders like `your_db_user`, `your_api_key`, etc.
  </action>
  <verify>
    <automated>node -e "
      const fs = require('fs');
      const env = fs.readFileSync('.env', 'utf8');
      const example = fs.readFileSync('.env.example', 'utf8');
      const sections = ['DATABASE', 'PORTAL', 'MAILING', 'PATHS', 'APP'];
      const requiredVars = ['USER', 'PASSWORD', 'SERVER', 'DATABASE', 'URL', 'TENANT_ID', 'API_KEY', 'API_SECRET', 'DATABASES', 'EXTERNAL_IDS', 'MAIL_TRANSPORT', 'PATH', 'PROVIDERS_PATH', 'LOG_PATH', 'IMPORT_CFDIS_ROUTE', 'ARG', 'NOMBRE', 'RFC', 'REGIMEN', 'TIMEZONE', 'AUTO_TERMINATE'];
      let ok = true;
      for (const s of sections) {
        if (!env.includes(s)) { console.error('Missing section in .env: ' + s); ok = false; }
        if (!example.includes(s)) { console.error('Missing section in .env.example: ' + s); ok = false; }
      }
      for (const v of requiredVars) {
        if (!env.includes(v + '=')) { console.error('Missing var in .env: ' + v); ok = false; }
        if (!example.includes(v + '=')) { console.error('Missing var in .env.example: ' + v); ok = false; }
      }
      if (ok) console.log('PASS: All sections and variables present');
      else process.exit(1);
    "
    </automated>
  </verify>

    - MAIL_TRANSPORT variable is present in both files
    - Real values from existing .env files are preserved in the unified .env

      const fs = require('fs');
      const path = require('path');
      let ok = true;
      // Legacy dir exists

      // Old files should NOT exist in root anymore (only the ones that existed)

      for (const f of oldFiles) {
        if (fs.existsSync(f)) { console.error('FAIL: old file still in root: ' + f); ok = false; }
