**1. Create `src/routes/schemas/po-schemas.js`:**

```javascript
const Joi = require('joi');
const config = require('../../config');
const tenantMax = Math.max(config.portal.tenants.length - 1, 0);
```

Schemas:

- `diagnosticSchema` (query): poNumber (string, required, trimmed), database (string, default 'COPDAT'), empresa (string, default 'COPDAT')
- `querySchema` (query): poNumbers (string or array of strings, required -- comma-separated string should be split or accept array), database (string, optional -- null default), tenantIndex (int, 0-max, default 0)
- `uploadSchema` (body): poNumbers (string or array of strings, required), database (string, optional), tenantIndex (int, 0-max, default 0)
- `updateSchema` (body): poNumber (string, required, trimmed), database (string, optional), tenantIndex (int, 0-max, default 0), dryRun (boolean, default true -- per user decision on destructive endpoints)
- `addressDiagnosticSchema` (query): poNumber (string, required, trimmed), database (string, default 'COPDAT')
- `paymentFormDiagnosticSchema` (query): poNumber (string, required, trimmed), database (string, optional)
- `uploadAuthorizedSchema` (body): tenantIndex (int, 0-max, default 0)
- `lifecycleSchema` (body): mode (string, required, valid('analyze', 'process', 'tenant')), tenantIndex (int, 0-max, default 0), database (string, optional)

For `poNumbers`: Use `Joi.alternatives().try(Joi.string(), Joi.array().items(Joi.string()))`. In the route handler, normalize to array: if string, split by comma. This handles both `?poNumbers=OC001,OC002` (query) and `{"poNumbers": ["OC001","OC002"]}` (body).

Export all schemas.

**2. Create `src/routes/po-routes.js`:**

```javascript
const express = require('express');
const router = express.Router();
const { validate } = require('../middleware/validate');
const { asyncHandler } = require('../middleware/async-handler');
const { sendResult } = require('../middleware/send-result');
const { rateLimit } = require('express-rate-limit');
```

Create a local writeLimiter (same config as in server.js: windowMs 1min, limit 10, envelope-shaped message) to avoid circular dependency with server.js.

8 routes:

a. `GET /diagnostic` -- validate(diagnosticSchema, 'query'), asyncHandler:
   ```javascript
   const { diagnosticPO } = require('../scripts/po-diagnostic');
   const result = await diagnosticPO(req.query.poNumber, req.query.database, req.query.empresa);
   sendResult(res, result);
   ```

b. `GET /query` -- validate(querySchema, 'query'), asyncHandler:
   ```javascript
   const { testSpecificPurchaseOrders } = require('../scripts/po-query');
   // Normalize poNumbers to array
   let poNumbers = req.query.poNumbers;
   if (typeof poNumbers === 'string') poNumbers = poNumbers.split(',').map(s => s.trim());
   const result = await testSpecificPurchaseOrders(poNumbers, req.query.database, req.query.tenantIndex);
   sendResult(res, result);
   ```

c. `POST /upload` -- validate(uploadSchema, 'body'), writeLimiter, asyncHandler:
   ```javascript
   const { uploadSpecificPurchaseOrders } = require('../scripts/po-upload');
   let poNumbers = req.body.poNumbers;
   if (typeof poNumbers === 'string') poNumbers = poNumbers.split(',').map(s => s.trim());
   const result = await uploadSpecificPurchaseOrders(poNumbers, req.body.database, req.body.tenantIndex);
   sendResult(res, result);
   ```

d. `PUT /update` -- validate(updateSchema, 'body'), writeLimiter, asyncHandler:
   ```javascript
   const { testPurchaseOrderUpdate } = require('../scripts/po-update');
   const result = await testPurchaseOrderUpdate(req.body.poNumber, req.body.database, req.body.tenantIndex, req.body.dryRun);
   sendResult(res, result);
   ```
   NOTE: dryRun defaults to true via Joi schema -- destructive endpoint safety net.

e. `GET /address-diagnostic` -- validate(addressDiagnosticSchema, 'query'), asyncHandler:
   ```javascript
   const { diagnosticPOAddress } = require('../scripts/po-address-diagnostic');
   const result = await diagnosticPOAddress(req.query.poNumber, req.query.database);
   sendResult(res, result);
   ```

f. `GET /payment-form-diagnostic` -- validate(paymentFormDiagnosticSchema, 'query'), asyncHandler:
   ```javascript
   const { diagnosticPaymentForm } = require('../scripts/po-payment-form-diagnostic');
   const result = await diagnosticPaymentForm(req.query.poNumber, req.query.database);
   sendResult(res, result);
   ```

g. `POST /upload-authorized` -- validate(uploadAuthorizedSchema, 'body'), writeLimiter, asyncHandler:
   ```javascript
   const { uploadAuthorizedPOs } = require('../scripts/upload-authorized-pos');
   const result = await uploadAuthorizedPOs({ tenantIndex: req.body.tenantIndex });
   sendResult(res, result);
   ```

h. `POST /lifecycle` -- validate(lifecycleSchema, 'body'), writeLimiter, asyncHandler:
   ```javascript
   const { analyzeOrders, processOrders, testTenant } = require('../scripts/test-order-lifecycle');
   const { mode, tenantIndex, database } = req.body;
   let result;
   switch (mode) {
       case 'analyze':
           result = await analyzeOrders({ tenantIndex, database });
           break;
       case 'process':
           result = await processOrders({ tenantIndex, database });
           break;
       case 'tenant':
           result = await testTenant({ tenantIndex });
           break;
   }
   sendResult(res, result);
   ```

Export router.
